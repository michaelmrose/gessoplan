% Shared running major-part header for standard and Florilegium PDFs.
% Pandoc renders top-level Org "* Part ..." headings as LaTeX \section.
% Only these section marks update the left mark; subsection titles do not.
\usepackage{fancyhdr}
\usepackage[fit]{truncate}

% Reserve space within the existing narrow top margin. Without includehead,
% the header can be clipped above the paper in the standard publication.
\setlength{\headheight}{12pt}
\setlength{\headsep}{8pt}
\geometry{includehead}

\renewcommand{\sectionmark}[1]{\markboth{#1}{}}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[R]{%
  \footnotesize
  \nouppercase{\truncate{0.79\headwidth}{\leftmark}}%
  \hspace{0.8em}\textbar\hspace{0.8em}\thepage}
\renewcommand{\headrulewidth}{0pt}
